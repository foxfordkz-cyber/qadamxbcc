```python
import asyncio
from collections import Counter
from app.ml.model import MODEL_PATH, load_model
from scripts.compare_phone_coercion import assisted_cases
from scripts.evaluate_fraud_model import evaluate_scenario

async def run():
    model = load_model(MODEL_PATH, allow_training=False)
    for variant in ("full_hybrid", "without_ml"):
        counts = Counter()
        for _, case in assisted_cases(seed=20261217):
            answer = await evaluate_scenario(case, variant, model)
            counts[(case.fraudulent, answer["status"])] += 1
        print(variant, counts)

asyncio.run(run())
```
